
# Email
[notifications:]
- provider: Email
  to:
    - user1@email.com
    - user2@email.com
  subject: 'Build {{status}}'                  # optional
  message: "{{message}}, {{commitId}}, ..."    # optional
  on_build_success: true|false
  on_build_failure: true|false
  on_build_status_changed: true|false

# HipChat
[notifications:]
- provider: HipChat
  auth_token:
    secure: <AUTH-TOKEN>
  room: ProjectA
  template: "{message}, {commitId}, ..."
  on_build_success: true|false
  on_build_failure: true|false
  on_build_status_changed: true|false
